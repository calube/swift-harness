@Test("adding the same item twice increments quantity — catches duplicate cart lines")
func addTwice() {
  var cart = Cart()
  cart.add(Item(id: 1))
  cart.add(Item(id: 1))
  #expect(cart._storageForTesting.count == 1)
  #expect(cart._storageForTesting.capacity >= 1)
  #expect(cart._mutationCountForTesting == 2)
}
